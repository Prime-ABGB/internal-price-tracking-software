CREATE TABLE historical_prices (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    sku TEXT NOT NULL,
	category TEXT NOT NULL,
    date TEXT NOT NULL,
    prime_price REAL,
	prime_stock VARCHAR,
    mdcomp_price REAL,
	mdcomp_stock VARCHAR,
    vedant_price REAL,
	vedant_stock VARCHAR,
	pcstudio_price REAL,
	pcstudio_stock VARCHAR,
	clarion_price REAL,
	clarion_stock VARCHAR,
	ehubs_price REAL,
	ehubs_stock VARCHAR,
    UNIQUE (sku, date)
);

UPDATE historical_prices AS current
SET 
    clarion_price = previous.clarion_price,
    clarion_stock = previous.clarion_stock
FROM (
    SELECT 
        sku, 
        clarion_price, 
        clarion_stock 
    FROM historical_prices 
    WHERE date = '2024-09-17'
) AS previous
WHERE 
    current.sku = previous.sku 
    AND current.date = '2024-09-18';
